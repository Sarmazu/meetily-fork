cmake_minimum_required(VERSION 3.19)
project(whisper_vulkan_regression LANGUAGES C CXX)

if(NOT EXISTS "${WHISPER_CPP_DIR}/ggml/include/ggml-vulkan.h")
    message(FATAL_ERROR "WHISPER_CPP_DIR must point to the locked sys crate's whisper.cpp")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(WIN32)
    include("${CMAKE_CURRENT_LIST_DIR}/../../.github/force-portable-ggml.cmake")
endif()
set(GGML_VULKAN ON CACHE BOOL "Exercise the Vulkan backend" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static regression executable" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
add_subdirectory("${WHISPER_CPP_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/whisper" EXCLUDE_FROM_ALL)
add_executable(whisper-vulkan-softmax softmax.cpp)
target_link_libraries(whisper-vulkan-softmax PRIVATE ggml ggml-base ggml-vulkan)
include(CTest)
foreach(case no-mask mask)
    add_test(NAME softmax-${case} COMMAND whisper-vulkan-softmax ${case})
    set_tests_properties(softmax-${case} PROPERTIES SKIP_RETURN_CODE 77)
endforeach()
